#include <ESP8266WiFi.h>
#include <espnow.h>

#define LED 1

String mens;

void setup() {
  WiFi.mode(WIFI_STA);
  if(esp_now_init() != 0) panico();

  esp_now_set_self_role(ESP_NOW_ROLE_SLAVE);
  esp_now_register_recv_cb(recep_hand);
}

void panico(){
  bool est = false;
  pinMode(LED,OUTPUT);
  while(true){
    digitalWrite(LED,est);
    est = !est;
    ESP.deepSleep(10e6);
  }
}
